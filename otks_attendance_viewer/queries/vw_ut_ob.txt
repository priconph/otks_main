SELECT ob.pkid as ob_pkid, ob.date_time_created,ob.created_by,ob.empno, CONCAT(hris.lastname,' ,',hris.firstname) as emp_name,hris.Position,hris.Department,ob.ob_date,if(ob.ob_purpose=0,'UT','OB') as ob_purpose,ob.ob_place,ob.ob_reason,ob.ob_time_out,ob.ob_time_return,ob.ob_approver,ob.supervisor_username,
if(ob.ob_status=0,'For Approval',(if(ob.ob_status=1,'Approved',(if(ob.ob_status=2,'Disapproved',(if(ob.ob_status=3,'Cancelled',(if(ob.ob_status=4,'For Edit','ERROR'))))))))) as ob_status_name	
 FROM `tkob` ob INNER JOIN db_hris.vw_EmpInfo_Rapid hris ON hris.EmpNo = ob.empno AND ob.logdel=0 ORDER BY ob.pkid DESC
